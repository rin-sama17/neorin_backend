داخل Admin چند Helper اضافه می‌کنیم.

public function hasRole($role)
{
    return $this->roles()
        ->where('slug',$role)
        ->exists();
}
public function hasPermission($permission)
{
    return $this->roles()
        ->whereHas('permissions',function($q) use ($permission){

            $q->where('slug',$permission);

        })
        ->exists();
}
